cmake_minimum_required(VERSION 3.24)
project(pen_stabilizer VERSION 0.1.0 LANGUAGES CXX)
add_library(pen_stabilizer INTERFACE)
add_library(pen_stabilizer::pen_stabilizer ALIAS pen_stabilizer)
target_include_directories(pen_stabilizer INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(pen_stabilizer INTERFACE cxx_std_17)
option(PEN_STABILIZER_TESTS "Build conformance tests and example" ${PROJECT_IS_TOP_LEVEL})
if(PEN_STABILIZER_TESTS)
    enable_testing()
    add_executable(conformance tests/conformance.cpp)
    target_link_libraries(conformance PRIVATE pen_stabilizer::pen_stabilizer)
    if(PENTRACE_REFERENCE)
        target_sources(conformance PRIVATE "${PENTRACE_REFERENCE}/src/core.cpp")
        target_include_directories(conformance PRIVATE "${PENTRACE_REFERENCE}/src")
        target_compile_definitions(conformance PRIVATE PENTRACE_REFERENCE=1)
        target_compile_features(conformance PRIVATE cxx_std_20)
    endif()
    add_test(NAME conformance COMMAND conformance)
    add_executable(ink_example examples/minimal.cpp)
    target_link_libraries(ink_example PRIVATE pen_stabilizer::pen_stabilizer)
    add_test(NAME example COMMAND ink_example)
    add_executable(ink_benchmark tests/benchmark.cpp)
    target_link_libraries(ink_benchmark PRIVATE pen_stabilizer::pen_stabilizer)
endif()
